// Classic worker that runs ffmpeg.wasm (single-threaded core) fully in the browser.
// Served by our own Worker (a cached copy of @ffmpeg/core 0.12.10), so no third party sees the visitor.
const CORE = self.location.origin + "/vendor/ffmpeg-core.js";
const WASM = CORE.replace(/\.js$/, ".wasm");
let ff = null;
let duration = 0;
let tail = [];

async function load() {
  if (ff) return ff;
  importScripts(CORE);
  ff = await self.createFFmpegCore({ mainScriptUrlOrBlob: CORE + "#" + btoa(JSON.stringify({ wasmURL: WASM })) });
  ff.setLogger(({ message }) => {
    const m = /Duration: (\d+):(\d+):(\d+\.\d+)/.exec(message || "");
    // The first input sets the length; a cover image reports its own, much shorter, duration after it.
    if (m && !duration) duration = +m[1] * 3600 + +m[2] * 60 + +m[3];
    tail.push(String(message || ""));
    if (tail.length > 40) tail.shift();
    self.postMessage({ type: "log", message });
  });
  ff.setProgress(({ progress, time }) => {
    // time is in microseconds of output; fall back to ffmpeg's own estimate.
    let p = duration > 0 && time > 0 ? time / 1e6 / duration : progress;
    if (!(p >= 0 && p <= 1)) p = 0;
    self.postMessage({ type: "progress", progress: p });
  });
  return ff;
}

self.onmessage = async ({ data }) => {
  // extra: further inputs such as a cover image, each with the input options that go before its -i.
  const { id, file, extra = [], args, outName } = data;
  const written = [];
  try {
    self.postMessage({ id, type: "status", status: "Loading ffmpeg, about 30 MB the first time …" });
    const core = await load();
    duration = 0;
    tail = [];
    const inName = "in_" + (file.name || "input").replace(/[^\w.]+/g, "_");
    core.FS.writeFile(inName, new Uint8Array(await file.arrayBuffer()));
    written.push(inName);
    const more = [];
    for (const [i, x] of extra.entries()) {
      const name = "extra" + i + "_" + (x.file.name || "input").replace(/[^\w.]+/g, "_");
      core.FS.writeFile(name, new Uint8Array(await x.file.arrayBuffer()));
      written.push(name);
      more.push(...(x.pre || []), "-i", name);
    }
    self.postMessage({ id, type: "status", status: "Converting …" });
    // Drop metadata and chapters (location, device, names) from every result.
    core.exec("-i", inName, ...more, "-map_metadata", "-1", "-map_chapters", "-1", ...args, outName);
    const ret = core.ret;
    core.reset();
    for (const name of written.splice(0)) core.FS.unlink(name);
    // A video without sound asked for audio, or audio without sound asked for a cover video.
    if (ret !== 0 && tail.some((l) => /does not contain any stream|matches no streams/.test(l))) throw new Error("noaudio");
    if (ret !== 0) throw new Error("ffmpeg exit code " + ret);
    const out = core.FS.readFile(outName);
    core.FS.unlink(outName);
    self.postMessage({ id, type: "done", data: out.buffer }, [out.buffer]);
  } catch (e) {
    for (const name of written) try { ff.FS.unlink(name); } catch {}
    self.postMessage({ id, type: "error", error: String(e && e.message ? e.message : e) });
  }
};
